document.querySelectorAll('[data-pay]').forEach(b=>b.addEventListener('click',()=>{const k=b.dataset.pay;const link=k==='deposit'?CONFIG.DEPOSIT_LINK:CONFIG.INVOICE_LINK;if(link){location.href=link;return}document.getElementById('payWhat').textContent=k==='deposit'?'Appointment deposit (credited to your bill)':'Pay an invoice from Marc';document.getElementById('payAmt').textContent=k==='deposit'?'Deposit':'Invoice balance';document.getElementById('invRow').style.display=k==='invoice'?'block':'none';modal.classList.add('on')}));
document.getElementById('payClose').onclick=()=>modal.classList.remove('on');modal.addEventListener('click',e=>{if(e.target===modal)modal.classList.remove('on')});document.addEventListener('keydown',e=>{if(e.key==='Escape')modal.classList.remove('on')});
</script>
